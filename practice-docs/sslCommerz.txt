Step 5: 
Create Payment Session

Controller:
---------
const SSLCommerzPayment = require('sslcommerz-lts');

const store_id = process.env.SSLC_STORE_ID;
const store_passwd = process.env.SSLC_STORE_PASSWORD;
const is_live = false;

const transactionId = `TXN-${Date.now()}`;

const data = {
    total_amount: order.totalPrice,
    currency: 'BDT',
    tran_id: transactionId,

    success_url:
      'http://localhost:5000/api/payment/success',

    fail_url:
      'http://localhost:5000/api/payment/fail',

    cancel_url:
      'http://localhost:5000/api/payment/cancel',

    ipn_url:
      'http://localhost:5000/api/payment/ipn',

    product_name: 'Order Payment',
    product_category: 'Ecommerce',
    product_profile: 'general',

    cus_name: user.name,
    cus_email: user.email,
    cus_phone: user.phone,
};

Step 6: 
Initialize Payment
const sslcz = new SSLCommerzPayment(
    store_id,
    store_passwd,
    is_live
);

const apiResponse =
    await sslcz.init(data);

res.json({
    paymentUrl: apiResponse.GatewayPageURL
});

Frontend:

window.location.href = paymentUrl;
Step 7: Success Route
router.post(
    '/success',
    paymentSuccessController
);

Example:

exports.paymentSuccessController =
async (req, res) => {

   const { tran_id } = req.body;

   const order =
      await Order.findOne({
          transactionId: tran_id
      });

   order.paymentStatus = 'paid';

   await order.save();

   res.redirect(
      'http://localhost:3000/payment-success'
   );
};
